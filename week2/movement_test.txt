function movePlayer(scene) {
    scene.actionManager = new BABYLON.ActionManager(scene);
    scene.inputMap = {};

    scene.actionManager.registerAction(new BABYLON.ExecuteCodeAction(BABYLON.ActionManager.OnKeyDownTrigger, (event) => {
        scene.inputMap[event.sourceEvent.key] = event.sourceEvent.type == "keydown";
    }));

    scene.actionManager.registerAction(new BABYLON.ExecuteCodeAction(BABYLON.ActionManager.OnKeyUpTrigger, (event) => {
        scene.inputMap[event.sourceEvent.key] = event.sourceEvent.type == "keydown";
    }));
};

function updateFromKeyboard(scene) {
    if (scene.inputMap["ArrowUp"]) {
        vertical = "UP";
    } else if (scene.inputMap["ArrowDown"]) {
        vertical = "DOWN";
    } else {
        vertical = "";
    }


    if (scene.inputMap["ArrowLeft"]) {
        horizontal = "LEFT";
    } else if (scene.inputMap["ArrowRight"]) {
        horizontal = "RIGHT";
    } else {
        horizontal = "";
    }
};
